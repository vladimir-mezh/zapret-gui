cmake_minimum_required(VERSION 3.20)
project(ZapretGUI VERSION 0.3.0 LANGUAGES CXX)
add_executable(ZapretGUI WIN32 src/main.cpp)
target_compile_features(ZapretGUI PRIVATE cxx_std_17)
target_compile_definitions(ZapretGUI PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX GUI_VERSION="${PROJECT_VERSION}")
target_link_libraries(ZapretGUI PRIVATE user32 gdi32 comctl32 uxtheme ole32 shell32 uuid winhttp bcrypt advapi32)
option(BUILD_MCP "Build optional standalone MCP server" OFF)
if(BUILD_MCP)
  add_executable(ZapretMCP src/mcp.cpp)
  target_compile_features(ZapretMCP PRIVATE cxx_std_17)
  target_compile_definitions(ZapretMCP PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(ZapretMCP PRIVATE ole32 shell32 uuid)
  if(MSVC)
    target_compile_options(ZapretMCP PRIVATE /utf-8 /W4)
    set_property(TARGET ZapretMCP PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_link_options(ZapretMCP PRIVATE -municode)
  endif()
endif()
if(MSVC)
  target_compile_options(ZapretGUI PRIVATE /utf-8 /W4)
  set_property(TARGET ZapretGUI PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(ZapretGUI PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  add_executable(backend_tests tests/backend_tests.cpp)
  add_executable(testing_tests tests/testing_tests.cpp)
  add_executable(test_process_fixture tests/test_process_fixture.cpp)
  foreach(test_target IN ITEMS core_tests backend_tests testing_tests test_process_fixture)
    target_compile_features(${test_target} PRIVATE cxx_std_17)
    target_compile_definitions(${test_target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(${test_target} PRIVATE shell32 ole32 uuid winhttp bcrypt advapi32)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /utf-8 /W4)
      set_property(TARGET ${test_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(NOT test_target STREQUAL "core_tests" AND NOT test_target STREQUAL "test_process_fixture")
      target_link_options(${test_target} PRIVATE -municode)
    endif()
  endforeach()
  add_test(NAME core COMMAND core_tests)
  add_test(NAME backend COMMAND backend_tests "${CMAKE_CURRENT_BINARY_DIR}/test-data/backend")
  add_test(NAME testing COMMAND testing_tests "${CMAKE_CURRENT_BINARY_DIR}/test-data/testing" $<TARGET_FILE:test_process_fixture>)
endif()

cmake_minimum_required(VERSION 3.20)
project(ZapretGUI VERSION 0.2.0 LANGUAGES CXX)
add_executable(ZapretGUI WIN32 src/main.cpp)
target_compile_features(ZapretGUI PRIVATE cxx_std_17)
target_compile_definitions(ZapretGUI PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX GUI_VERSION="${PROJECT_VERSION}")
target_link_libraries(ZapretGUI PRIVATE user32 gdi32 comctl32 uxtheme ole32 shell32 uuid winhttp bcrypt advapi32)
option(BUILD_MCP "Build optional standalone MCP server" OFF)
if(BUILD_MCP)
  add_executable(ZapretMCP src/mcp.cpp)
  target_compile_features(ZapretMCP PRIVATE cxx_std_17)
  target_compile_definitions(ZapretMCP PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(ZapretMCP PRIVATE ole32 shell32 uuid)
  if(MSVC)
    target_compile_options(ZapretMCP PRIVATE /utf-8 /W4)
    set_property(TARGET ZapretMCP PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_link_options(ZapretMCP PRIVATE -municode)
  endif()
endif()
if(MSVC)
  target_compile_options(ZapretGUI PRIVATE /utf-8 /W4)
  set_property(TARGET ZapretGUI PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(ZapretGUI PRIVATE -municode)
endif()

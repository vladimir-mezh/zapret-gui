cmake_minimum_required(VERSION 3.20)
project(ZapretMCP VERSION 0.3.0 LANGUAGES CXX)
set(MCP_RELEASE_VERSION "${PROJECT_VERSION}" CACHE STRING "Independent MCP release version")
if(NOT MCP_RELEASE_VERSION MATCHES "^[0-9][A-Za-z0-9._-]*$")
  message(FATAL_ERROR "Invalid MCP release version")
endif()
add_executable(ZapretMCP src/main.cpp)
add_executable(ZapretMcpLauncher src/launcher.cpp)
target_compile_features(ZapretMcpLauncher PRIVATE cxx_std_17)
target_compile_definitions(ZapretMcpLauncher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ZapretMcpLauncher PRIVATE ole32 shell32 uuid)
target_compile_features(ZapretMCP PRIVATE cxx_std_17)
target_compile_definitions(ZapretMCP PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX MCP_VERSION="${MCP_RELEASE_VERSION}")
target_link_libraries(ZapretMCP PRIVATE ole32 shell32 uuid)
if(MSVC)
  target_compile_options(ZapretMcpLauncher PRIVATE /utf-8 /W4)
  set_property(TARGET ZapretMcpLauncher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ZapretMCP PRIVATE /utf-8 /W4)
  set_property(TARGET ZapretMCP PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(ZapretMcpLauncher PRIVATE -municode)
  target_link_options(ZapretMCP PRIVATE -municode)
endif()
